def migrate_placement_test_schema() -> None:
    """Apply additive SQLite migrations for auth, curriculum, and review sessions."""
    inspector = inspect(engine)
    tables = set(inspector.get_table_names())
    with engine.begin() as connection:
        if "users" in tables:
            user_columns = {column["name"] for column in inspector.get_columns("users")}
            if "role" not in user_columns:
                connection.execute(
                    text("ALTER TABLE users ADD COLUMN role VARCHAR NOT NULL DEFAULT 'USER'")
                )
            if "status" not in user_columns:
                connection.execute(
                    text("ALTER TABLE users ADD COLUMN status VARCHAR NOT NULL DEFAULT 'ACTIVE'")
                )
            if "full_name" not in user_columns:
                connection.execute(text("ALTER TABLE users ADD COLUMN full_name VARCHAR"))
            if "gender" not in user_columns:
                connection.execute(text("ALTER TABLE users ADD COLUMN gender VARCHAR"))
            if "bio" not in user_columns:
                connection.execute(text("ALTER TABLE users ADD COLUMN bio VARCHAR"))
            if "avatar_key" not in user_columns:
                connection.execute(
                    text("ALTER TABLE users ADD COLUMN avatar_key VARCHAR NOT NULL DEFAULT 'bunny'")
                )

        if "lessons" in tables:
            lesson_columns = {column["name"] for column in inspector.get_columns("lessons")}
            if "pronunciation_passing_score" not in lesson_columns:
                connection.execute(
                    text("ALTER TABLE lessons ADD COLUMN pronunciation_passing_score INTEGER NOT NULL DEFAULT 70")
                )
            if "pronunciation_max_attempts" not in lesson_columns:
                connection.execute(
                    text("ALTER TABLE lessons ADD COLUMN pronunciation_max_attempts INTEGER NOT NULL DEFAULT 7")
                )

        if "words" in tables:
            word_columns = {column["name"] for column in inspector.get_columns("words")}
            if "ipa" not in word_columns:
                connection.execute(text("ALTER TABLE words ADD COLUMN ipa VARCHAR"))
            if "reference_audio_path" not in word_columns:
                connection.execute(text("ALTER TABLE words ADD COLUMN reference_audio_path VARCHAR"))

        if "pronunciation_attempts" in tables:
            pronunciation_columns = {
                column["name"]
                for column in inspector.get_columns("pronunciation_attempts")
            }
            if "lesson_id" not in pronunciation_columns:
                connection.execute(
                    text("ALTER TABLE pronunciation_attempts ADD COLUMN lesson_id INTEGER")
                )
            if "overall_score" not in pronunciation_columns:
                connection.execute(
                    text("ALTER TABLE pronunciation_attempts ADD COLUMN overall_score INTEGER NOT NULL DEFAULT 0")
                )
            if "character_results" not in pronunciation_columns:
                connection.execute(
                    text("ALTER TABLE pronunciation_attempts ADD COLUMN character_results JSON")
                )
            if "phoneme_results" not in pronunciation_columns:
                connection.execute(
                    text("ALTER TABLE pronunciation_attempts ADD COLUMN phoneme_results JSON")
                )

        if "levels" in tables:
            level_columns = {column["name"] for column in inspector.get_columns("levels")}
            new_columns = {
                "placement_test_enabled": "INTEGER NOT NULL DEFAULT 0",
                "placement_test_question_count": "INTEGER NOT NULL DEFAULT 10",
                "placement_test_passing_score": "INTEGER NOT NULL DEFAULT 7",
                "is_active": "INTEGER NOT NULL DEFAULT 1",
            }
            for name, definition in new_columns.items():
                if name not in level_columns:
                    connection.execute(text(f"ALTER TABLE levels ADD COLUMN {name} {definition}"))

        if "topics" in tables:
            topic_columns = {column["name"] for column in inspector.get_columns("topics")}
            if "quick_review_limit" not in topic_columns:
                connection.execute(
                    text("ALTER TABLE topics ADD COLUMN quick_review_limit INTEGER NOT NULL DEFAULT 15")
                )

        if "review_session_items" in tables:
            review_item_columns = {
                column["name"]
                for column in inspector.get_columns("review_session_items")
            }
            if "question_data" not in review_item_columns:
                connection.execute(
                    text("ALTER TABLE review_session_items ADD COLUMN question_data JSON")
                )
            if "correct_answer" not in review_item_columns:
                connection.execute(
                    text("ALTER TABLE review_session_items ADD COLUMN correct_answer TEXT")
                )
            if "answer" not in review_item_columns:
                connection.execute(
                    text("ALTER TABLE review_session_items ADD COLUMN answer TEXT")
                )

        for table_name in ("topics", "lessons", "words"):
            if table_name in tables:
                columns = {
                    column["name"]
                    for column in inspector.get_columns(table_name)
                }
                if "is_active" not in columns:
                    connection.execute(
                        text(
                            f"ALTER TABLE {table_name} "
                            "ADD COLUMN is_active INTEGER NOT NULL DEFAULT 1"
                        )
                    )


def initialize_database() -> None:
    migrate_placement_test_schema()

from fastapi import Depends, HTTPException, status
from sqlalchemy.orm import Session
from fastapi.security import HTTPBearer
from app.database import SessionLocal
from models.user import User
from services.token_service import decode_access_token

bearer_scheme = HTTPBearer()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


def get_current_user(
    credentials = Depends(bearer_scheme),
    db: Session = Depends(get_db)
) -> User:
    token = credentials.credentials
    credentials_error = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )

    try:
        payload = decode_access_token(token)
        user_id = int(payload["sub"])
    except (ValueError, TypeError, KeyError):
        raise credentials_error

    user = (
        db.query(User)
        .filter(User.user_id == user_id)
        .first()
    )

    if user is None or user.status != "ACTIVE":
        raise credentials_error

    return user


def require_admin(
    current_user: User = Depends(get_current_user),
) -> User:
    if current_user.role != "ADMIN":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Admin access required",
        )
    return current_user

from fastapi import APIRouter, Depends, HTTPException, status
from pydantic import BaseModel, Field
from sqlalchemy.orm import Session

from app.database import SessionLocal
from app.security import get_current_user
from models.user import User
from schemas.auth import LoginRequest, TokenResponse
from services.auth_service import (
    authenticate_user,
    create_access_token,
)


router = APIRouter(prefix="/auth", tags=["Authentication"])


def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


@router.post("/login", response_model=TokenResponse)
def login(
    login_data: LoginRequest,
    db: Session = Depends(get_db),
):
    try:
        user = authenticate_user(
            db=db,
            identifier=login_data.identifier,
            password=login_data.password,
        )
    except ValueError as error:
        message = str(error)

        if message == "Account is locked or inactive":
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=message,
            ) from error

        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid username/email or password",
            headers={"WWW-Authenticate": "Bearer"},
        ) from error

    access_token = create_access_token(user)

    return TokenResponse(access_token=access_token)

        "bio": user.bio or "",

        "created_at": user.created_at,
    }

@router.put("/me")
def update_current_user_profile(
    data: ProfileUpdate,
    user: User = Depends(get_current_user),
    db: Session = Depends(get_db),

    full_name = data.full_name.strip()
    if not full_name:
        raise HTTPException(
            status_code=422,

from fastapi import APIRouter, Depends, File, Form, HTTPException, UploadFile, status
from fastapi.responses import FileResponse
from sqlalchemy.orm import Session

from app.security import get_current_user, get_db
from models.curriculum import Word
from models.user import User
from services.pronunciation_service import assess_recording, get_reference_audio


router = APIRouter(prefix="/pronunciation", tags=["Pronunciation"])


@router.post("/assess")
def assess_pronunciation(
    word_id: int = Form(...),
    lesson_id: int = Form(...),
    target_text: str = Form(...),
    audio: UploadFile = File(...),
    db: Session = Depends(get_db),
    user: User = Depends(get_current_user),
):
    try:
        audio_bytes = audio.file.read()
        if len(audio_bytes) > 10 * 1024 * 1024:
            raise HTTPException(status_code=413, detail="Audio is too large")
        extension = (audio.filename or "recording.webm").rsplit(".", 1)[-1].lower()
        if extension not in {"webm", "wav", "ogg", "m4a", "mp3"}:
            extension = "webm"
        return assess_recording(
            db,
            user,
            word_id=word_id,
            lesson_id=lesson_id,
            target_text=target_text,
            audio_bytes=audio_bytes,
            extension=extension,
        )
    except ValueError as error:
        raise HTTPException(status_code=422, detail=str(error)) from error


@router.get("/words/{word_id}/reference-audio")
async def reference_audio(
    word_id: int,
    db: Session = Depends(get_db),
    _: User = Depends(get_current_user),
):
    word = db.query(Word).filter_by(word_id=word_id, is_active=1).first()
    if word is None:
        raise HTTPException(status_code=404, detail="Word not found")
    try:
        path = await get_reference_audio(word)
        db.commit()
    except Exception as error:
        raise HTTPException(status_code=502, detail="Could not generate reference audio") from error
    return FileResponse(path, media_type="audio/mpeg", filename=f"{word.word}.mp3")

from pathlib import Path
import sqlite3


# Thư mục gốc của project
PROJECT_ROOT = Path(__file__).resolve().parents[2]

# Đường dẫn đến file SQL và database SQLite
SQL_FILE = PROJECT_ROOT / "database" / "database.sql"
DATABASE_FILE = PROJECT_ROOT / "backend" / "database.db"


def init_database():
    if not SQL_FILE.exists():
        raise FileNotFoundError(
            f"Không tìm thấy file SQL: {SQL_FILE}"
        )

    if DATABASE_FILE.exists():
        raise FileExistsError(
            f"Database đã tồn tại: {DATABASE_FILE}. "
            "Không khởi tạo lại để tránh ghi đè dữ liệu."
        )

    connection = sqlite3.connect(DATABASE_FILE)

    try:
        connection.execute("PRAGMA foreign_keys = ON")

        sql_script = SQL_FILE.read_text(encoding="utf-8")
        connection.executescript(sql_script)

        connection.commit()
        print("Khởi tạo database thành công!")

    except Exception:
        connection.rollback()
        connection.close()

        # Xóa database tạo dở khi khởi tạo thất bại
        if DATABASE_FILE.exists():
            DATABASE_FILE.unlink()

        raise

    else:
        connection.close()


if __name__ == "__main__":
    init_database()

    *,

    email: str,

    role: str,
    password_env: str,

    user = db.query(User).filter_by(email=email).first()
    if user is not None:
        if user.role != role or user.status != "ACTIVE":
            raise RuntimeError(f"Existing account {email} is not an active {role} account")
        return user, False

    password = os.environ.get(password_env)
    if not password or len(password) < 12:
        password = "AdminDemo2026!" if role == "ADMIN" else "UserDemo2026!"
    user = User(
        username=username,
        email=email,
        full_name=full_name,
        password_hash=password_hasher.hash(password),
        role=role,
        status="ACTIVE",
    )
    db.add(user)
    db.flush()
    return user, True

def authenticate_user(
    db: Session,
    identifier: str,
    password: str,
) -> User:
    user = (
        db.query(User)
        .filter(
            (User.username == identifier)
            | (User.email == identifier)
        )
        .first()
    )

    if user is None:
        raise ValueError("Invalid username/email or password")

    if not password_hasher.verify(password, user.password_hash):
        raise ValueError("Invalid username/email or password")

    if user.status != "ACTIVE":

    return user


def create_access_token(user: User) -> str:
    now = datetime.now(timezone.utc)

    payload = {
        "sub": str(user.user_id),
        "role": user.role,
        "iat": now,
        "exp": now + timedelta(
            minutes=ACCESS_TOKEN_EXPIRE_MINUTES
        ),
    }

    return jwt.encode(
        payload,
        SECRET_KEY,
        algorithm=ALGORITHM,
    )

    "dit", "dkm", "dmm", "dcm", "dm", "dcmm", "vcl", "vkl", "vl", "clgt", "cc", "cl", "lon", "buoi", "cac", "deo", "du",
}
# Từ bắt đầu bằng các gốc này cũng bị chặn (vd: fuuuck, shitload, motherfuckers)
BANNED_STEMS = ("fuck", "fuk", "fuuu", "motherf", "shit", "bitch", "cunt", "nigg", "fagg", "whor", "slut", "dickhead")


def _plain(token: str) -> str:
    """Bỏ dấu tiếng Việt, chữ thường, chỉ giữ a-z 0-9 * và dấu nháy: "Địt" -> "dit"."""
    text = unicodedata.normalize("NFD", token.lower())
    text = "".join(ch for ch in text if unicodedata.category(ch) != "Mn")
    text = text.replace("đ", "d")
    return re.sub(r"[^a-z0-9*']", "", text)


def allowed_words(target: str) -> set[str]:
    """Các từ có trong từ / câu mẫu thì không bao giờ bị chặn."""
    return {_plain(w).replace("'", "") for w in re.split(r"[\s-]+", target or "") if _plain(w)}


def is_banned(token: str, allowed: set[str] | None = None) -> bool:
    word = _plain(token)
    if not word:
        return False
    if "*" in word:  # Chrome tự che từ tục thành "f***"
        return True
    bare = word.replace("'", "")
    if allowed and bare in allowed:
        return False
    return bare in BANNED_WORDS or bare.startswith(BANNED_STEMS)


def clean_text(text: str, target: str = "") -> tuple[str, int]:
    """Xoá từ cấm khỏi câu. Trả về (câu đã lọc, số từ bị bỏ qua)."""
    allowed = allowed_words(target)
    kept, removed = [], 0
    for token in (text or "").split():
        if is_banned(token, allowed):
            removed += 1
        else:
            kept.append(token)
    return " ".join(kept), removed

def level_is_unlocked(db: Session, user: User, level: Level) -> bool:
    if user.role == "ADMIN" or not level.placement_test_enabled:
        return True

    passed_higher_test = (
        db.query(PlacementTestAttempt.attempt_id)
        .join(Level, Level.level_id == PlacementTestAttempt.level_id)
        .filter(
            PlacementTestAttempt.user_id == user.user_id,
            PlacementTestAttempt.passed == 1,
            or_(
                Level.display_order > level.display_order,
                and_(
                    Level.display_order == level.display_order,
                    Level.level_id >= level.level_id,
                ),
            ),
        )
        .first()
    )
    return passed_higher_test is not None

def create_attempt(db: Session, user_id: int, level_id: int) -> PlacementTestAttempt:
    level = get_level_or_raise(db, level_id)
    if not level.placement_test_enabled:
        raise ValueError("This level does not have an entrance test")

    words = get_level_words(db, level_id)
    if not words:
        raise ValueError("This level has no active vocabulary assigned to its lessons")

    unique_meanings = {_normalize(word.meaning) for word in words if _normalize(word.meaning)}
    if len(unique_meanings) < 4:
        raise ValueError("This level needs at least four distinct meanings to create multiple-choice questions")

    selected_words = random.sample(words, min(level.placement_test_question_count, len(words)))
    passing_score = min(level.placement_test_passing_score, len(selected_words))

    attempt = PlacementTestAttempt(
        user_id=user_id,
        level_id=level_id,
        total_questions=len(selected_words),
        passing_score=passing_score,
    )
    db.add(attempt)
    try:
        db.flush()
        for word in selected_words:
            correct_meaning = word.meaning.strip()
            distractors_by_meaning = {}
            for candidate in words:
                meaning = candidate.meaning.strip()
                normalized = _normalize(meaning)
                if candidate.word_id != word.word_id and normalized != _normalize(correct_meaning):
                    distractors_by_meaning.setdefault(normalized, meaning)
            if len(distractors_by_meaning) < 3:
                raise ValueError(
                    "This level does not have enough distinct distractor meanings for every question"
                )
            distractors = random.sample(list(distractors_by_meaning.values()), 3)
            options = [correct_meaning, *distractors]
            random.shuffle(options)
            db.add(
                PlacementTestAttemptItem(
                    attempt_id=attempt.attempt_id,
                    word_id=word.word_id,
                    question_data={
                        "word": word.word,
                        "pronunciation": word.pronunciation,
                        "options": options,
                    },
                )
            )
        db.commit()
        db.refresh(attempt)
        return attempt
    except Exception:
        db.rollback()
        raise


def submit_answer(
    db: Session,
    user_id: int,
    attempt_id: int,
    item_id: int,
    answer: str,
) -> PlacementTestAttemptItem:
    attempt = (
        db.query(PlacementTestAttempt)
        .filter_by(attempt_id=attempt_id, user_id=user_id)
        .first()
    )
    if attempt is None:
        raise LookupError("Placement test attempt not found")
    if attempt.completed_at is not None:
        raise ValueError("Placement test attempt is already completed")

    item = (
        db.query(PlacementTestAttemptItem)
        .filter_by(item_id=item_id, attempt_id=attempt_id)
        .first()
    )
    if item is None:
        raise LookupError("Question not found in this attempt")
    if item.is_answered:
        raise ValueError("This question has already been answered")

    normalized_answer = _normalize(answer)
    options = item.question_data["options"]
    if normalized_answer not in {_normalize(option) for option in options}:
        raise ValueError("Answer must match one of the provided options")

    word = db.query(Word).filter_by(word_id=item.word_id).first()
    if word is None:
        raise LookupError("Question vocabulary no longer exists")

    item.answer = answer
    item.is_answered = 1
    item.is_correct = int(normalized_answer == _normalize(word.meaning))
    item.answered_at = datetime.now(timezone.utc).replace(tzinfo=None)
    try:
        db.commit()
        db.refresh(item)
        return item
    except Exception:
        db.rollback()
        raise


def complete_attempt(
    db: Session,
    user_id: int,
    attempt_id: int,
) -> PlacementTestAttempt:
    attempt = (
        db.query(PlacementTestAttempt)
        .filter_by(attempt_id=attempt_id, user_id=user_id)
        .first()
    )
    if attempt is None:
        raise LookupError("Placement test attempt not found")
    if attempt.completed_at is not None:
        return attempt

    items = (
        db.query(PlacementTestAttemptItem)
        .filter_by(attempt_id=attempt_id)
        .all()
    )
    if len(items) != attempt.total_questions or any(not item.is_answered for item in items):
        raise ValueError("Answer every question before completing the placement test")

    correct_answers = sum(item.is_correct for item in items)
    attempt.correct_answers = correct_answers
    attempt.passed = int(correct_answers >= attempt.passing_score)
    attempt.completed_at = datetime.now(timezone.utc).replace(tzinfo=None)
    try:
        db.commit()
        db.refresh(attempt)
        return attempt
    except Exception:
        db.rollback()
        raise

def get_learning_streak(db: Session, user: User) -> dict:
    rows = (
        db.query(UserDailyActivity.activity_date)
        .filter(UserDailyActivity.user_id == user.user_id)
        .order_by(UserDailyActivity.activity_date.desc())
        .all()
    )
    active_dates = sorted({row.activity_date for row in rows if row.activity_date})
    if not active_dates:
        return {
            "current_streak": 0,
            "longest_streak": 0,
            "total_active_days": 0,
            "activity_dates": [],
        }

    longest = 1
    running = 1
    for previous, current in zip(active_dates, active_dates[1:]):
        if (current - previous).days == 1:
            running += 1
            longest = max(longest, running)
        else:
            running = 1

    today = date.today()
    latest = active_dates[-1]
    current = 0
    if (today - latest).days in (0, 1):
        cursor = latest
        active_set = set(active_dates)
        while cursor in active_set:
            current += 1
            cursor = cursor.fromordinal(cursor.toordinal() - 1)

    return {
        "current_streak": current,
        "longest_streak": longest,
        "total_active_days": len(active_dates),
        "activity_dates": [item.isoformat() for item in active_dates[-60:]],
    }

def mark_word_learned(

    )

def _meaning_distractors(
    words: list[Word],
    candidates: list[Word],
) -> list[str]:
    correct_meaning = next((word.meaning for word in words if word.meaning), "")
    correct_normalized = _normalize(correct_meaning)
    unique_meanings = {}
    for word in [*words, *candidates]:
        meaning = (word.meaning or "").strip()
        normalized = _normalize(meaning)
        if normalized and normalized != correct_normalized:
            unique_meanings.setdefault(normalized, meaning)
    return list(unique_meanings.values())


def _word_distractors(words: list[Word], candidates: list[Word]) -> list[str]:
    correct_words = {_normalize(word.word) for word in words}
    unique_words = {}
    for word in [*words, *candidates]:
        value = (word.word or "").strip()
        normalized = _normalize(value)
        if normalized and normalized not in correct_words:
            unique_words.setdefault(normalized, value)
    return list(unique_words.values())


def _blank_sentence(word: Word) -> str:
    example = (word.example or "").strip()
    if example:
        pattern = re.compile(
            rf"(?<!\w){re.escape(word.word)}(?!\w)",
            flags=re.IGNORECASE,
        )
        sentence, count = pattern.subn("____", example, count=1)
        if count:
            return sentence
    return f"I learned the word ____ today."


def _build_exercise(
    word: Word,
    topic_words: list[Word],
    all_words: list[Word],
) -> tuple[dict, str]:
    meanings = _meaning_distractors([word], [*topic_words, *all_words])
    word_options = _word_distractors([word], [*topic_words, *all_words])
    available = ["meaning_input"]
    if len(meanings) >= 3:
        available.append("meaning_choice")
    if len(word_options) >= 3:
        available.append("fill_blank")
    kind = random.choice(available)

    if kind == "meaning_choice":
        options = random.sample(meanings, 3) + [word.meaning.strip()]
        random.shuffle(options)
        return (
            {
                "kind": kind,
                "word": word.word,
                "pronunciation": word.pronunciation,
                "options": options,
            },
            word.meaning.strip(),
        )
    if kind == "fill_blank":
        options = random.sample(word_options, 3) + [word.word.strip()]
        random.shuffle(options)
        return (
            {
                "kind": kind,
                "sentence": _blank_sentence(word),
                "options": options,
            },
            word.word.strip(),
        )
    return (
        {
            "kind": kind,
            "word": word.word,
            "pronunciation": word.pronunciation,
        },
        word.meaning.strip(),
    )


def create_quick_review(db: Session, user: User, topic_id: int) -> dict:
    topic = _accessible_topic(db, user, topic_id)
    learned_words = _learned_words(db, user.user_id, topic_id)
    if not learned_words:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Learn at least one word in this topic before starting a quick review",
        )

    selected_words = random.sample(
        learned_words,
        min(topic.quick_review_limit, len(learned_words)),
    )
    topic_words = _topic_words(db, topic_id)
    all_words = _user_available_words(db, user)

    session = ReviewSession(
        user_id=user.user_id,
        topic_id=topic_id,
        review_type="NORMAL",
        total_items=len(selected_words),
        completed_items=0,
    )
    db.add(session)
    try:
        db.flush()
        response_items = []
        for word in selected_words:
            question_data, correct_answer = _build_exercise(word, topic_words, all_words)
            item = ReviewSessionItem(
                review_session_id=session.review_session_id,
                word_id=word.word_id,
                question_data=question_data,
                correct_answer=correct_answer,
            )
            db.add(item)
            db.flush()
            response_items.append(
                {
                    "item_id": item.session_item_id,
                    "word_id": word.word_id,
                    "question_data": question_data,
                }
            )
        db.commit()
        db.refresh(session)
    except Exception:
        db.rollback()
        raise

    return {
        "review_session_id": session.review_session_id,
        "topic_id": topic_id,
        "total_items": len(response_items),
        "items": response_items,
    }


def submit_quick_review_answer(
    db: Session,
    user: User,
    review_session_id: int,
    item_id: int,
    answer: str,
) -> dict:
    review_session = (
        db.query(ReviewSession)
        .filter_by(review_session_id=review_session_id, user_id=user.user_id)
        .first()
    )
    if review_session is None:
        raise HTTPException(status_code=404, detail="Review session not found")

    item = (
        db.query(ReviewSessionItem)
        .filter_by(session_item_id=item_id, review_session_id=review_session_id)
        .first()
    )
    if item is None:
        raise HTTPException(status_code=404, detail="Review question not found")

    if item.is_answered:
        correct_answer = item.correct_answer or ""
        return {
            "item_id": item.session_item_id,
            "is_correct": bool(item.is_correct),
            "correct_answer": correct_answer,
            "completed_items": review_session.completed_items,
            "total_items": review_session.total_items,
            "session_completed": review_session.completed_at is not None,
        }

    question_data = item.question_data or {}
    normalized_answer = _normalize(answer)
    if question_data.get("kind") in {"meaning_choice", "fill_blank"}:
        options = question_data.get("options", [])
        if normalized_answer not in {_normalize(option) for option in options}:
            raise HTTPException(
                status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
                detail="Answer must match one of the provided options",
            )

    is_correct = normalized_answer == _normalize(item.correct_answer or "")
    now = _now()
    item.answer = answer
    item.is_answered = 1
    item.is_correct = int(is_correct)
    item.answered_at = now

    word_progress = (
        db.query(UserWordProgress)
        .filter_by(user_id=user.user_id, word_id=item.word_id)
        .first()
    )
    if word_progress is None:
        raise HTTPException(status_code=409, detail="Learned-word progress no longer exists")
    word_progress.review_count += 1
    word_progress.correct_count += int(is_correct)
    word_progress.wrong_count += int(not is_correct)
    word_progress.last_reviewed_at = now

    topic_progress = (
        db.query(UserTopicProgress)
        .filter_by(user_id=user.user_id, topic_id=review_session.topic_id)
        .first()
    )
    if topic_progress is None:
        topic_progress = UserTopicProgress(
            user_id=user.user_id,
            topic_id=review_session.topic_id,
            learned_count=len(_learned_words(db, user.user_id, review_session.topic_id)),
        )
        db.add(topic_progress)
    topic_progress.last_reviewed_at = now
    _record_daily_activity(db, user.user_id, review_count=1)

    review_session.completed_items += 1
    if review_session.completed_items == review_session.total_items:
        review_session.completed_at = now

    try:
        db.commit()
    except Exception:
        db.rollback()
        raise

    return {
        "item_id": item.session_item_id,
        "is_correct": is_correct,
        "correct_answer": item.correct_answer or "",
        "completed_items": review_session.completed_items,
        "total_items": review_session.total_items,
        "session_completed": review_session.completed_at is not None,
    }

_ASSESSOR = None

# Model AI chấm phát âm (~1.26GB, tải từ HuggingFace lần đầu rồi lưu ở ~/.cache/huggingface).
# _LOAD_LOCK: nhiều request cùng lúc chỉ nạp model 1 lần (trước đây mỗi request tự tải -> tải trùng).
# _ASSESS_LOCK: model dùng chung giữ trạng thái câu mẫu (set_sentence), nên chấm lần lượt từng bản ghi.
_LOAD_LOCK = threading.Lock()
_ASSESS_LOCK = threading.Lock()


def _get_assessor():
    global _ASSESSOR
    if _ASSESSOR is None:
        with _LOAD_LOCK:
            if _ASSESSOR is None:
                from pronounce_assess import PronounceAssessModel

                _ASSESSOR = PronounceAssessModel()
    return _ASSESSOR


def preload_assessor_in_background() -> None:
    """Nạp sẵn model ở luồng nền khi backend khởi động,
    để người học bấm chấm lần đầu không phải chờ tải model."""
    def _load():
        try:
            _get_assessor()
            print("[pronunciation] Model chấm phát âm đã sẵn sàng.")
        except Exception as error:  # không làm sập server nếu thiếu mạng / thiếu thư viện
            print(f"[pronunciation] Chưa nạp được model: {error}")

    threading.Thread(target=_load, name="preload-pronounce-model", daemon=True).start()


def decode_audio(audio_bytes: bytes, extension: str = "webm") -> np.ndarray:
    _CACHE_DIR.mkdir(parents=True, exist_ok=True)
    with tempfile.NamedTemporaryFile(suffix=f".{extension}", delete=False) as source:
        source.write(audio_bytes)
        source_path = source.name

    wav_path = source_path + ".wav"
    try:
        import imageio_ffmpeg

        ffmpeg = imageio_ffmpeg.get_ffmpeg_exe()
        attempts = (
            [
                ffmpeg, "-v", "error", "-y",
                "-i", source_path,
                "-vn", "-ac", "1", "-ar", "16000",
                "-f", "wav", wav_path,
            ],
            [
                ffmpeg, "-v", "error", "-y",
                "-fflags", "+genpts",
                "-i", source_path,
                "-vn", "-ac", "1", "-ar", "16000",
                "-f", "wav", wav_path,
            ],
        )
        result = None
        detail = ""
        for command in attempts:
            if Path(wav_path).exists():
                Path(wav_path).unlink()
            result = subprocess.run(
                command,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                check=False,
            )
            if result.returncode == 0 and Path(wav_path).is_file() and Path(wav_path).stat().st_size > 44:
                break
            detail = result.stderr.decode("utf-8", "replace").strip()
        if result.returncode != 0:
            detail = result.stderr.decode("utf-8", "replace").strip()
            raise ValueError(
                "Could not decode the recorded audio"
                + (f": {detail[-300:]}" if detail else "")
            )
        audio, wav_sample_rate = sf.read(wav_path, dtype="float32", always_2d=False)
        if wav_sample_rate != 16000:
            raise ValueError("Decoded audio sample rate is invalid")
        if hasattr(audio, "mean") and getattr(audio, "ndim", 1) > 1:
            audio = audio.mean(axis=1)
        if len(audio) == 0:
            raise ValueError("Recorded audio contains no usable signal")
        return audio.astype(np.float32)
    except ValueError:
        raise
    except Exception as error:
        raise ValueError(f"Could not decode the recorded audio: {type(error).__name__}: {error}") from error
    finally:
        try:
            Path(source_path).unlink(missing_ok=True)
            wav_candidate = Path(source_path + ".wav")
            wav_candidate.unlink(missing_ok=True)
        except Exception:
            pass


def _chunks(audio: np.ndarray, chunk_size: int = 8000) -> Iterable[np.ndarray]:
    for start in range(0, len(audio), chunk_size):
        chunk = audio[start:start + chunk_size]
        if len(chunk):
            yield chunk.astype(np.float32)


# Mức độ của từng trạng thái (số lớn hơn = nặng hơn), dùng khi nhiều âm rơi vào cùng 1 chữ
_SEVERITY = {"correct": 0, "close": 1, "missing": 2, "wrong": 3}
# gop (0..1) của âm đọc lệch: từ ngưỡng này trở lên coi là "gần đúng" thay vì "sai"
_CLOSE_GOP = 0.5


def _phoneme_status(event: dict) -> str:
    label = event.get("label")
    if label == "omitted":
        return "missing"
    if label == "mispronounced":
        gop = event.get("gop")
        try:
            if gop is not None and math.isfinite(float(gop)) and float(gop) >= _CLOSE_GOP:
                return "close"
        except (TypeError, ValueError):
            pass
        return "wrong"
    return "correct"


def _letter_statuses(target: str, phoneme_events: list[dict]) -> list[str]:
    """Trạng thái cho từng ký tự của câu mẫu.

    Âm vị thứ p (trong tổng n âm của câu mẫu) được gán vào chữ cái tương ứng theo tỉ lệ
    vị trí, chỉ tính chữ cái/số (bỏ khoảng trắng, dấu câu) nên không tô đỏ khoảng trắng.
    """
    chars = list(target)
    statuses = ["correct"] * len(chars)
    letter_idx = [i for i, ch in enumerate(chars) if ch.isalnum()]
    ref_events = [e for e in phoneme_events if e.get("position") is not None]
    if not letter_idx or not ref_events:
        return statuses

    n_ref = max(int(e["position"]) for e in ref_events) + 1
    for event in ref_events:
        status = _phoneme_status(event)
        if status == "correct":
            continue
        pos = int(event["position"])
        k = min(len(letter_idx) - 1, int((pos + 0.5) * len(letter_idx) / n_ref))
        char_index = letter_idx[k]
        if _SEVERITY[status] > _SEVERITY[statuses[char_index]]:
            statuses[char_index] = status
    return statuses


def _character_results(target: str, phoneme_events: list[dict]) -> list[dict]:
    target = target.strip()
    if not target:
        return []
    statuses = _letter_statuses(target, phoneme_events)
    return [{"character": char, "status": status} for char, status in zip(target, statuses)]


def _word_results(target: str, character_results: list[dict]) -> list[dict]:
    """Gộp kết quả từng chữ cái thành kết quả từng từ (dùng cho chế độ đọc cả câu)."""
    words, current = [], []
    for item in character_results + [{"character": " ", "status": "correct"}]:
        if item["character"].isspace():
            if current:
                text = "".join(c["character"] for c in current)
                letters = [c["status"] for c in current if c["character"].isalnum()]
                bad = [s for s in letters if s != "correct"]
                if not bad:
                    status = "correct"
                elif all(s == "missing" for s in bad) and len(bad) * 2 >= len(letters):
                    status = "missing"
                elif all(s == "close" for s in bad) or len(bad) <= max(1, len(letters) // 4):
                    status = "close"
                else:
                    status = "wrong"
                words.append({"text": text, "status": status})
                current = []
        else:
            current.append(item)
    return words


def _issues(phoneme_events: list[dict], limit: int = 5) -> list[dict]:
    """Các âm cần sửa: đọc lệch (âm mẫu -> âm máy nghe được) hoặc bị bỏ sót."""
    issues = []
    for event in phoneme_events:
        if event.get("position") is None:
            continue
        status = _phoneme_status(event)
        if status == "correct":
            continue
        issues.append({
            "status": status,
            "expected": event.get("phoneme"),
            "heard": event.get("decoded") if status != "missing" else None,
        })
        if len(issues) >= limit:
            break
    return issues


def assess_recording(
    db: Session,
    user: User,
    word_id: int,
    lesson_id: int,
    target_text: str,
    audio_bytes: bytes,
    extension: str,
) -> dict:
    word = db.query(Word).filter_by(word_id=word_id, is_active=1).first()
    if word is None:
        raise ValueError("Word not found")

    lesson = db.query(Lesson).filter_by(lesson_id=lesson_id, is_active=1).first()
    if lesson is None:
        raise ValueError("Lesson not found")

    target = (target_text or word.word).strip()
    audio = decode_audio(audio_bytes, extension)
    if len(audio) < 1600:
        raise ValueError("Recorded audio is too short")

    assessor = _get_assessor()
    events = []
    prosody = {}
    with _ASSESS_LOCK:
        assessor.set_sentence(target)
        for event in assessor.stream_decode(_chunks(audio), sample_rate=16000):
            if event.label == "prosody":
                prosody = {
                    "monotony_score": event.monotony_score,
                    "rhythm_score": event.rhythm_score,
                    "boundary_score": event.boundary_score,
                    "speaking_rate": event.speaking_rate,
                }
                continue
            events.append({
                "phoneme": event.phoneme,
                "position": event.position,
                "label": event.label,
                "decoded": event.decoded,
                "gop": event.gop,
            })

    if not events:
        raise ValueError("No pronunciation phonemes could be detected")

    weights = {"correct": 1.0, "mispronounced": 0.35, "omitted": 0.0, "insertion": 0.10}
    reference_events = [event for event in events if event["position"] is not None]
    phoneme_score = sum(weights.get(event["label"], 0.0) for event in reference_events)
    if reference_events:
        phoneme_score = round(100 * phoneme_score / len(reference_events))
    else:
        phoneme_score = 0

    gop_values = [
        float(event["gop"])
        for event in reference_events
        if event.get("gop") is not None and math.isfinite(float(event["gop"]))
    ]
    gop_score = round(100 * sum(gop_values) / len(gop_values)) if gop_values else phoneme_score
    overall = round(0.75 * phoneme_score + 0.25 * gop_score)

    character_results = _character_results(target, events)

    attempt = PronunciationAttempt(
        user_id=user.user_id,
        word_id=word.word_id,
        lesson_id=lesson.lesson_id,
        word=word.word.strip().lower(),
        mode="word",
        target_text=target,
        transcript="",
        score=overall,
        overall_score=overall,
        character_results=character_results,
        phoneme_results=events,
    )
    db.add(attempt)
    db.commit()
    db.refresh(attempt)

    attempts = (
        db.query(PronunciationAttempt)
        .filter_by(user_id=user.user_id, lesson_id=lesson.lesson_id, word_id=word.word_id)
        .count()
    )
    passed = overall >= lesson.pronunciation_passing_score
    exhausted = attempts >= lesson.pronunciation_max_attempts

    return {
        "attempt_id": attempt.attempt_id,
        "word": word.word,
        "target_text": target,
        "score": overall,
        "passed": passed,
        "attempt": attempts,
        "max_attempts": lesson.pronunciation_max_attempts,
        "passing_score": lesson.pronunciation_passing_score,
        "advance": passed or exhausted,
        "characters": character_results,
        "words": _word_results(target, character_results),
        "issues": _issues(events),
        "phonemes": events,
        "prosody": prosody,
    }


async def get_reference_audio(word: Word, force: bool = False) -> Path:
    cache_dir = _CACHE_DIR
    cache_dir.mkdir(parents=True, exist_ok=True)
    cache_path = cache_dir / f"word_{word.word_id}.mp3"
    if force and cache_path.exists():
        cache_path.unlink()
    if not cache_path.exists():
        communicate = edge_tts.Communicate(
            text=word.word,
            voice="en-US-JennyNeural",
        )
        await communicate.save(str(cache_path))
    word.reference_audio_path = str(cache_path)
    return cache_path


def ensure_reference_audio(word: Word, force: bool = False) -> Path:

from jose import JWTError, jwt

from services.auth_service import SECRET_KEY, ALGORITHM


def decode_access_token(token: str) -> dict:
    try:
        payload = jwt.decode(
            token,
            SECRET_KEY,
            algorithms=[ALGORITHM],
        )

        # Token phải có subject là user_id.
        subject = payload.get("sub")

        if subject is None:
            raise ValueError("Invalid token")

        return payload

    except JWTError as error:
        raise ValueError("Invalid or expired token") from error

from pwdlib import PasswordHash
from sqlalchemy.orm import Session
import re
from models.user import User

password_hasher = PasswordHash.recommended()

def get_all_users(db: Session) -> list[User]:
    return db.query(User).order_by(User.user_id).all()

def create_user(
    db: Session,
    username: str | None,
    email: str,
    password: str,
    full_name: str | None = None,
    gender: str | None = None,
    avatar_key: str = "bunny",
) -> User:
    if not username:
        base = re.sub(r"[^A-Za-z0-9_]", "", email.split("@", 1)[0])[:24] or "user"
        if len(base) < 3:
            base = f"{base}user"[:24]
        username = base
        suffix = 1
        while db.query(User.user_id).filter(User.username == username).first():
            tail = str(suffix)
            username = f"{base[:30 - len(tail)]}{tail}"
            suffix += 1

    existing_username = (
        db.query(User)
        .filter(User.username == username)
        .first()
    )

    if existing_username:
        raise ValueError("Username already exists")

    existing_email = (
        db.query(User)
        .filter(User.email == email)
        .first()
    )

    if existing_email:
        raise ValueError("Email already exists")

    hashed_password = password_hasher.hash(password)

    new_user = User(
        username=username,
        email=email,
        password_hash=hashed_password,
        full_name=full_name,
        gender=gender,
        avatar_key=avatar_key,
    )

    db.add(new_user)

    try:
        db.commit()
        db.refresh(new_user)
    except Exception:
        db.rollback()
        raise

    return new_user
